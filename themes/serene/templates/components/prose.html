{# Prose components. Replaces the pre-0.23 macros/prose.html and the callout/quote/ detail/mermaid/youtube/figure shortcodes. Everything except `back_link` is invoked from markdown content, so the emitted HTML is handed to the markdown parser: keep lines flush-left and emit no blank lines, otherwise 4-space indentation becomes an indented code block and a blank line ends the raw HTML block. Hence the `{%-` whitespace-trimming control tags below. #} {% component back_link(path, text = "Back") %}
← {{ text }}
{% endcomponent back_link %} {% component callout(name, title = "") %} {%- set icon = load_data(path="icon/" ~ name ~ ".svg") %}

{{ body | markdown | safe }}
{%- else %}
{{ icon | safe }}
{{ body | markdown | safe }}
{%- endif %}
{% endcomponent callout %} {% component note(title = "") %}{% %}{{ body }}{% %}{% endcomponent note %} {% component tip(title = "") %}{% %}{{ body }}{% %}{% endcomponent tip %} {% component important(title = "") %}{% %}{{ body }}{% %}{% endcomponent important %} {% component warning(title = "") %}{% %}{{ body }}{% %}{% endcomponent warning %} {% component caution(title = "") %}{% %}{{ body }}{% %}{% endcomponent caution %} {% component quote(cite = "") %} {%- set icon = load_data(path="icon/quote.svg") %}
{{ icon | safe }}
{{ body | markdown | safe }}
{%- if cite %}
{{ "— " ~ cite | markdown | safe }}
{%- endif %}
{% endcomponent quote %} {% component detail(title, default_open = false) %}

{% endcomponent detail %} {% component mermaid() %}
